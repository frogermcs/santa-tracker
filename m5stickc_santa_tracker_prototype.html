// M5StickC Plus2 - Kids Santa Tracker (3 Screens)
// Libraries: M5StickCPlus2 (via Arduino Library Manager)
#include <M5StickCPlus2.h>

M5Canvas canvas(&StickCP2.Display);

int currentScreen = 0;   // 0: Side Anim, 1: Dashboard, 2: World Radar
int santaStep = 0;       // 0: Flying, 1: Chimney Drop, 2: Cookie Break, 3: Turbo
uint32_t kms = 418920;
uint32_t gifts = 1842900;
int waypointIdx = 3;

struct Waypoint { const char* name; int x; int y; };
Waypoint route[] = {
  {"NORTH POLE", 120, 18}, {"ROVANIEMI", 128, 28}, {"WARSAW, PL", 125, 42},
  {"KRAKOW, PL", 124, 46}, {"LONDON, UK", 110, 43}, {"CAIRO, EG", 132, 62},
  {"TOKYO, JP", 198, 52},  {"SYDNEY, AU", 204, 105}, {"NEW YORK", 68, 48}
};

void setup() {
  auto cfg = M5.config();
  StickCP2.begin(cfg);
  StickCP2.Display.setRotation(1); // 240x135 Landscape
  canvas.createSprite(240, 135);
}

void loop() {
  StickCP2.update();

  // Button A switches between the 3 screens
  if (StickCP2.BtnA.wasPressed()) {
    currentScreen = (currentScreen + 1) % 3;
    StickCP2.Speaker.tone(1200, 40);
  }
  // Button B cycles Santa's 4 side actions or jumps city
  if (StickCP2.BtnB.wasPressed()) {
    santaStep = (santaStep + 1) % 4;
    waypointIdx = (waypointIdx + 1) % 9;
    StickCP2.Speaker.tone(1800, 40);
  }

  // Increment live counters
  kms += random(5, 19);
  gifts += random(120, 650);

  canvas.fillSprite(TFT_BLACK);
  if (currentScreen == 0) drawSideSanta();
  else if (currentScreen == 1) drawDashboard();
  else drawWorldRadar();

  canvas.pushSprite(0, 0);
  delay(50);
}

void drawSideSanta() {
  // Sky & Snowflakes
  canvas.fillSprite(0x0814);
  for(int i=0; i<20; i++) canvas.drawPixel((i*37 + millis()/30)%240, (i*19 + millis()/20)%110, TFT_WHITE);
  
  // Header Bar
  canvas.fillRect(0, 0, 240, 18, 0x9000);
  canvas.setTextColor(TFT_WHITE);
  canvas.setTextSize(1);
  canvas.setCursor(6, 5);
  const char* titles[] = {"1/4 FLYING SLEIGH", "2/4 CHIMNEY DROP", "3/4 COOKIE BREAK", "4/4 HYPER-WARP!"};
  canvas.print(titles[santaStep]);

  int bob = sin(millis() / 150.0) * 3;
  // Sleigh & Rudolph Side Profile
  canvas.fillRoundRect(75, 58 + bob, 52, 20, 4, TFT_RED);
  canvas.drawFastHLine(68, 80 + bob, 65, TFT_GOLD); // Sleigh runners
  canvas.fillRect(92, 46 + bob, 14, 14, TFT_RED);   // Santa body
  canvas.fillRect(94, 42 + bob, 10, 6, TFT_WHITE);  // Beard
  
  // Reindeer in front (Right side)
  canvas.fillRect(150, 56 - bob, 22, 12, 0x8200);
  canvas.fillCircle(174, 56 - bob, 3, TFT_RED);     // Rudolph glowing nose
}

void drawDashboard() {
  canvas.fillRect(0, 0, 240, 20, 0x03E0);
  canvas.setTextColor(TFT_WHITE);
  canvas.setCursor(8, 6);
  canvas.print("SANTA TELEMETRY DASHBOARD");

  canvas.setTextColor(TFT_YELLOW);
  canvas.setCursor(8, 30); canvas.print("DISTANCE (KM):");
  canvas.setTextSize(2); canvas.setCursor(8, 42); canvas.printf("%lu KM", kms);

  canvas.setTextSize(1); canvas.setTextColor(TFT_CYAN);
  canvas.setCursor(8, 68); canvas.print("GIFTS DELIVERED:");
  canvas.setTextSize(2); canvas.setCursor(8, 80); canvas.printf("%lu", gifts);

  canvas.setTextSize(1); canvas.setTextColor(TFT_GREEN);
  canvas.setCursor(8, 106); canvas.print("NEXT DESTINATION:");
  canvas.setTextColor(TFT_WHITE);
  canvas.setCursor(118, 106); canvas.print(route[waypointIdx].name);
}

void drawWorldRadar() {
  canvas.fillSprite(0x0022);
  // Grid lines
  for(int x=0; x<240; x+=30) canvas.drawFastVLine(x, 0, 135, 0x0165);
  for(int y=0; y<135; y+=27) canvas.drawFastHLine(0, y, 240, 0x0165);
  // Pulsing Santa Radar Dot
  Waypoint wp = route[waypointIdx];
  int r = (millis() / 120) % 10;
  canvas.drawCircle(wp.x, wp.y, r, TFT_RED);
  canvas.fillCircle(wp.x, wp.y, 3, TFT_RED);
}
